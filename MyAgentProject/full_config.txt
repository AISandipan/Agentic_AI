# Step 8 — NOW introduce LangGraph

This is where I would start changing your architecture.

```text

```

Your first graph can be:

```text
START
  ↓

  ↓
```

Example:

```python
from typing import TypedDict

from langgraph.graph import StateGraph, START, END

from ai.llm import llm
from ai.prompts import HEXBOT_PROMPT


class ChatState(TypedDict):
    user_message: str
    response: str


def chat_node(state: ChatState):

    chain = HEXBOT_PROMPT | llm

    result = chain.invoke({
        "input": state["user_message"]
    })

    return {
        "response": result.content
    }


builder = StateGraph(ChatState)

builder.add_node("chat", chat_node)

builder.add_edge(START, "chat")
builder.add_edge("chat", END)

graph = builder.compile()


def ask_hexbot(message: str):

    result = graph.invoke({
        "user_message": message,
        "response": ""
    })

    return result["response"]
```

Now:

```python
reply = ask_hexbot(msg)
```

replaces:

```python
reply = chatbot.get_response(msg)
```

```python
from langchain_community.document_loaders import DirectoryLoader
```

Then:

```python
loader = DirectoryLoader(
    "data/documents",
    glob="**/*.txt"
)

```

Split:

```python
splitter = RecursiveCharacterTextSplitter(
    chunk_size=800,
    chunk_overlap=100
)

chunks = splitter.split_documents(documents)
```

Create embeddings:

```python
embeddings = OpenAIEmbeddings(
    model="text-embedding-3-small"
)
```

Store:

```python
vectorstore = Chroma.from_documents(
    documents=chunks,
    embedding=embeddings,
    persist_directory="vectorstore"
)
```

Now your documents are searchable.

```python
from typing import TypedDict

from langgraph.graph import StateGraph, START, END
from langchain_ollama import ChatOllama


# --------------------------------------------------
# 1. State
# --------------------------------------------------

class ChatState(TypedDict):
    user_message: str
    response: str


# --------------------------------------------------
# 2. Local LLM
# --------------------------------------------------

llm = ChatOllama(
    model="llama3.2:3b",
    temperature=0.7
)


# --------------------------------------------------
# 3. Chat node
# --------------------------------------------------

def chat_node(state: ChatState):

    user_message = state["user_message"]

    response = llm.invoke(user_message)

    return {
        "response": response.content
    }


# --------------------------------------------------
# 4. Create graph
# --------------------------------------------------

builder = StateGraph(ChatState)

builder.add_node("chat", chat_node)

builder.add_edge(START, "chat")
builder.add_edge("chat", END)


# --------------------------------------------------
# 5. Compile graph
# --------------------------------------------------

graph = builder.compile()


# --------------------------------------------------
# 6. Function for our application
# --------------------------------------------------

def ask_graph(message: str):

    result = graph.invoke({
        "user_message": message,
        "response": ""
    })

    return result["response"]


# --------------------------------------------------
# 7. Test
# --------------------------------------------------

if __name__ == "__main__":

    answer = ask_graph(
        "Hello! Explain what you can do in one sentence."
    )

    print("\nHEXBot:")
    print(answer)
```

```python
from pathlib import Path

from langchain_community.document_loaders import TextLoader
from langchain_text_splitters import RecursiveCharacterTextSplitter
from langchain_ollama import OllamaEmbeddings
from langchain_chroma import Chroma


# --------------------------------------------------
# Paths
# --------------------------------------------------

BASE_DIR = Path(__file__).resolve().parent
DOCUMENTS_DIR = BASE_DIR / "documents"
CHROMA_DIR = BASE_DIR / "chroma_db"


# --------------------------------------------------
# Embedding model
# --------------------------------------------------

embeddings = OllamaEmbeddings(
    model="nomic-embed-text"
)


# --------------------------------------------------
# Load documents
# --------------------------------------------------

documents = []

for file_path in DOCUMENTS_DIR.glob("*.txt"):

    loader = TextLoader(
        str(file_path),
        encoding="utf-8"
    )

    documents.extend(loader.load())


print(f"Loaded {len(documents)} document(s).")


# --------------------------------------------------
# Split documents
# --------------------------------------------------

text_splitter = RecursiveCharacterTextSplitter(
    chunk_size=500,
    chunk_overlap=50
)

chunks = text_splitter.split_documents(documents)

print(f"Created {len(chunks)} chunk(s).")


# --------------------------------------------------
# Create Chroma database
# --------------------------------------------------

vectorstore = Chroma.from_documents(
    documents=chunks,
    embedding=embeddings,
    persist_directory=str(CHROMA_DIR),
    collection_name="hexbot_knowledge"
)


print("RAG database created successfully.")
print(f"Location: {CHROMA_DIR}")
```

```python
llm = ChatOllama(

)

    """
You are HEXBot, a helpful AI assistant.

Answer the user's question using the provided context.

If the answer cannot be found in the context, say:
"I don't have enough information in my documents to answer that."

Do not invent information.

Context:

{question}

Answer:

def ask_rag(question: str):
    retriever = get_retriever()

    documents = retriever.invoke(question)

    context = "\n\n".join(
        document.page_content
        for document in documents
    )

    messages = prompt.format_messages(
        context=context,
        question=question,
    )

    response = llm.invoke(messages)

    return response.content


if __name__ == "__main__":
```

```python
from typing import TypedDict

from langgraph.graph import StateGraph, START, END

from rag.rag_prompt import ask_rag


class ChatState(TypedDict):
    question: str
    answer: str


def rag_node(state: ChatState):
    question = state["question"]

    answer = ask_rag(question)

    return {
        "answer": answer
    }


# Create graph
builder = StateGraph(ChatState)

builder.add_node("rag", rag_node)

builder.add_edge(START, "rag")
builder.add_edge("rag", END)

graph = builder.compile()


if __name__ == "__main__":
    print("Testing LangGraph + RAG + Ollama...\n")

    question = "What technologies does HEXBot use?"

    result = graph.invoke({
        "question": question,

    })
```

```python
from typing import TypedDict

from langgraph.graph import StateGraph, START, END

from rag.rag_prompt import ask_rag


class ChatState(TypedDict):
    question: str
    answer: str


def rag_node(state: ChatState):
    question = state["question"]

    answer = ask_rag(question)

    return {
        "answer": answer
    }


# Create graph
builder = StateGraph(ChatState)

builder.add_node("rag", rag_node)

builder.add_edge(START, "rag")
builder.add_edge("rag", END)

graph = builder.compile()


# Function used by the GUI
def ask_graph(question: str) -> str:
    result = graph.invoke({
        "question": question,
        "answer": ""
    })

    return result["answer"]


if __name__ == "__main__":
    print("Testing LangGraph + RAG + Ollama...\n")

    question = "What technologies does HEXBot use?"

    result = graph.invoke({
        "question": question,
        "answer": ""
    })

    print("Question:")
    print(question)

    print("\nHEXBot:")
    print(result["answer"])
```
